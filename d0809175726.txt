@app.route("/api/sales", methods=["GET"])
def get_sales():
    conn = get_db()
    q = "SELECT * FROM sales WHERE 1=1"
    p = []
    if request.args.get("from"):
        q += " AND date>=?"; p.append(request.args["from"])
    if request.args.get("to"):
        q += " AND date<=?"; p.append(request.args["to"])
    q += " ORDER BY id DESC"
    sales = [row_to_dict(r) for r in conn.execute(q, p).fetchall()]
    for s in sales:
        s["items"] = [row_to_dict(r) for r in conn.execute("SELECT * FROM sale_items WHERE sale_id=?", (s["id"],)).fetchall()]
    conn.close()
    return jsonify(sales)
