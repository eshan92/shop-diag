@app.route("/api/stock-in", methods=["POST"])
def add_stock_in():
    data = request.json
    pid = int(data["product_id"]); qty = int(data["qty"]); cost = float(data["cost"])
    conn = get_db()
    prod = conn.execute("SELECT * FROM products WHERE id=?", (pid,)).fetchone()
    if not prod:
        conn.close()
        return jsonify({"ok": False}), 404
    conn.execute("UPDATE products SET stock=stock+?, buy_price=? WHERE id=?", (qty, cost, pid))
    conn.execute("INSERT INTO stock_in (date,product_id,product_name,qty,cost_per_unit,total_cost,supplier) VALUES (?,?,?,?,?,?,?)",
        (today(), pid, prod["name"], qty, cost, qty*cost, data.get("supplier","")))
    conn.commit()
    conn.close()
    return jsonify({"ok": True})
