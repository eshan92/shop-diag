// ===== MOBILE NAV =====
function toggleMenu(){document.querySelector('.sidebar').classList.toggle('open');}
function setBotNav(el){
  document.querySelectorAll('.bot-nav-item').forEach(function(b){b.classList.remove('active');});
  if(el)el.classList.add('active');
}
function toggleMore(e){e.stopPropagation();document.getElementById('bot-more').classList.toggle('open');}
function closeMore(){document.getElementById('bot-more').classList.remove('open');}
document.addEventListener('click',function(){closeMore();});

// ===== ADMIN NAV =====
function go(page){
  document.querySelectorAll('.page').forEach(function(p){p.classList.remove('active');});
  document.querySelectorAll('.nav-item').forEach(function(n){n.classList.remove('active');});
  var el=document.getElementById('page-'+page);
  if(el)el.classList.add('active');
  document.querySelectorAll('.nav-item').forEach(function(n){
    if(n.getAttribute('onclick')&&n.getAttribute('onclick').indexOf(page)>-1)n.classList.add('active');
  });
  if(page==='dashboard')rDash();
  if(page==='new-sale')rNewSale();
  if(page==='lots')rLots();
  if(page==='add-lot')rAddLot();
  if(page==='stock')rStock();
  if(page==='stock-in')rStockIn();
  if(page==='expenses')rExpenses();
