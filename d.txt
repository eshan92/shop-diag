function rStock(){
  api('GET','/api/products').then(function(prods){
    document.getElementById('scards').innerHTML=prods.map(function(p){
      var low=p.stock<=p.min_stock;
      return '<div class="sc '+(low?'red':'green')+'"><div class="si">'+(low?'⚠️':'✅')+'</div><div class="sl">'+p.name+'</div><div class="sv">'+p.stock+'</div><div class="su">'+p.unit+'s · Rs.'+fmt(p.sell_price)+'</div></div>';
    }).join('');
    document.getElementById('stktbl').innerHTML=prods.map(function(p){
      var low=p.stock<=p.min_stock;
      return '<tr><td class="bo">'+p.name+'</td><td>'+p.unit+'</td><td class="mn">Rs.'+fmt(p.buy_price)+'</td><td class="mn gr">Rs.'+fmt(p.sell_price)+'</td><td class="mn '+(low?'rd':'gr')+'">'+p.stock+'</td><td class="mn">Rs.'+fmt(p.stock*p.buy_price)+'</td><td><span class="badge '+(low?'br':'bg')+'">'+(low?'Low':'OK')+'</span></td></tr>';
    }).join('');
  });
}

function rStockIn(){
  api('GET','/api/products').then(function(prods){
    var sel=document.getElementById('sip');
    sel.innerHTML='<option value="">-- Select --</option>'+prods.map(function(p){return '<option value="'+p.id+'">'+p.name+' (Stock: '+p.stock+')</option>';}).join('');
  });
  api('GET','/api/stock-in').then(function(hist){
    document.getElementById('shist').innerHTML=hist.slice(0,15).map(function(h){
      return '<tr><td class="mn" style="font-size:10px">'+h.date+'</td><td>'+h.product_name+'</td><td class="mn gr">'+h.qty+'</td><td class="mn">Rs.'+fmt(h.total_cost)+'</td></tr>';
    }).join('')||'<tr><td colspan="4" style="text-align:center;color:var(--text3);padding:12px">No history</td></tr>';
  });
  var qe=document.getElementById('siq'),ce=document.getElementById('sic'),pe=document.getElementById('sipp');
  function upv(){var q=parseFloat(qe.value)||0,c=parseFloat(ce.value)||0;pe.textContent=q&&c?'Total: Rs.'+fmt(q*c):'';}
  qe.oninput=ce.oninput=upv;
}
