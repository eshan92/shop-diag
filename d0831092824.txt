def fifo_consume(conn, pid, qty, fallback=0.0, batches_out=None):
    """Take qty from the oldest batches. Returns total cost consumed.

    If `batches_out` is a list, append (batch_id, qty_taken) for each batch
    actually drawn from, so the caller can record exactly where the units
    came from (needed to reverse a void into the right batch later).
    """
    need = qty
    cost = 0.0
    rows = conn.execute(
        "SELECT id,remaining,buy_price FROM stock_batches "
        "WHERE product_id=? AND remaining>0 ORDER BY id", (pid,)
    ).fetchall()
    for b in rows:
        if need <= 0:
            break
        take = min(need, b["remaining"])
        cost += take * b["buy_price"]
        conn.execute("UPDATE stock_batches SET remaining=remaining-? WHERE id=?",
                     (take, b["id"]))
        if batches_out is not None:
            batches_out.append((b["id"], take))
        need -= take
    if need > 0:                      # batches short of real stock
        cost += need * fallback
    _sync_avg(conn, pid, fallback)
    return round(cost, 4)


def fifo_return(conn, pid, qty):
    """Put qty back into the newest consumed batches (used when voiding a sale).

    This is a best-effort guess used only as a fallback for sale_items that
    predate batch-level tracking (see fifo_return_precise). It does not know
    which batch the units actually came from.
    """
    need = qty
    rows = conn.execute(
        "SELECT id,qty,remaining,buy_price FROM stock_batches "
        "WHERE product_id=? ORDER BY id DESC", (pid,)
    ).fetchall()
    for b in rows:
        if need <= 0:
            break
        room = b["qty"] - b["remaining"]
        if room <= 0:
            continue
        give = min(need, room)
        conn.execute("UPDATE stock_batches SET remaining=remaining+? WHERE id=?",
                     (give, b["id"]))
        need -= give
    _sync_avg(conn, pid)


def fifo_return_precise(conn, pid, sale_item_id, qty):
    """Return stock to the exact batches a sale_item was drawn from, using
    the sale_item_batches record written at consume time. Falls back to the
    old newest-batch guess for sale_items recorded before this tracking
    existed (no rows found)."""
    rows = conn.execute(
        "SELECT batch_id, qty FROM sale_item_batches WHERE sale_item_id=?",
        (sale_item_id,),
    ).fetchall()
    if not rows:
        fifo_return(conn, pid, qty)
        return
    for r in rows:
        conn.execute("UPDATE stock_batches SET remaining=remaining+? WHERE id=?",
                     (r["qty"], r["batch_id"]))
    _sync_avg(conn, pid)

@bp.route("/api/batch/list")

    out = []
    for p in prods:
        bs = conn.execute(
            "SELECT id,date,qty,buy_price,remaining,supplier FROM stock_batches "
            "WHERE product_id=? ORDER BY id", (p["id"],)
        ).fetchall()

            "id": p["id"], "name": p["name"], "unit": p["unit"],

            "value": round(rem_val, 2),
            "batches": [dict(b) for b in bs],
        })
    conn.close()
