@app.route("/api/lots", methods=["POST"])
def add_lot():
    data = request.json
    buy = float(data["buy"])
    conn = get_db()
    count = conn.execute("SELECT COUNT(*) FROM lots WHERE buy=?", (buy,)).fetchone()[0]
    suffix = chr(65 + count)
    conn.execute("INSERT INTO lots (buy,suffix,sell,pieces,min_pieces,note,date) VALUES (?,?,?,?,?,?,?)",
        (buy, suffix, float(data["sell"]), int(data["pieces"]), int(data.get("min_pieces",50)), data.get("note",""), today()))
    conn.commit()
    lot = row_to_dict(conn.execute("SELECT * FROM lots ORDER BY id DESC LIMIT 1").fetchone())
    conn.close()
    return jsonify({"ok": True, "lot": lot})
