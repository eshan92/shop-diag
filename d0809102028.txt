@app.route("/api/products", methods=["POST"])
def add_product():
    data = request.json
    conn = get_db()
    conn.execute("INSERT INTO products (name,unit,buy_price,sell_price,stock,min_stock) VALUES (?,?,?,?,?,?)",
        (data["name"], data.get("unit","pcs"), float(data.get("buy_price",0)), float(data.get("sell_price",0)), int(data.get("stock",0)), int(data.get("min_stock",50))))
    conn.commit()
    conn.close()
    return jsonify({"ok": True})

@app.route("/api/products/<int:pid>", methods=["DELETE"])
def delete_product(pid):
    conn = get_db()
    conn.execute("DELETE FROM products WHERE id=?", (pid,))
    conn.commit()
    conn.close()
    return jsonify({"ok": True})

@app.route("/api/products/<int:pid>", methods=["PUT"])
def update_product(pid):
    data = request.json
    conn = get_db()
    for field in ["sell_price","buy_price","stock"]:
        if field in data:
            conn.execute("UPDATE products SET " + field + "=? WHERE id=?", (data[field], pid))
    conn.commit()
    conn.close()
    return jsonify({"ok": True})
