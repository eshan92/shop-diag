@app.route('/botv33/api/state')
def bot_state_v33():
    try:
        with open(BOT_STATE_FILE) as f: state = _json.load(f)
        prices = get_bot_prices()
        upnl, slots_a, slots_b = calc_upnl(state, prices)
        rpnl = sum(c.get('realized_pnl',0) for c in state.get('coins',{}).values())
        trades = []
        for sym, c in state.get('coins',{}).items():
            coin = sym.replace('USDT','')
            for t in c.get('closed_trades',[]):
                t2 = dict(t); t2['coin']=coin; trades.append(t2)
        trades.sort(key=lambda x: x.get('time',''), reverse=True)
        bal = sum(c.get('bal_a',0)+c.get('bal_b',0) for c in state.get('coins',{}).values())
        return jsonify({
            'realized_pnl': round(rpnl,2),
            'unrealized_pnl': round(upnl,2),
            'wallet': round(bal,2),
            'a_slots': slots_a,
            'b_slots': slots_b,
            'closed_trades': trades[:100],
        })
    except Exception as e:
        return jsonify({'error': str(e)}), 500
