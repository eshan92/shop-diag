=== EXPENSES API ===
@app.route("/api/expenses", methods=["GET"])
def get_expenses():
    conn = get_db()
    exp = [row_to_dict(r) for r in conn.execute("SELECT * FROM expenses ORDER BY id DESC LIMIT 50").fetchall()]
    conn.close()
    return jsonify(exp)

@app.route("/api/expenses", methods=["POST"])
def add_expense():
    data = request.json
    conn = get_db()
    conn.execute("INSERT INTO expenses (date,category,amount,description) VALUES (?,?,?,?)",
        (today(), data["category"], float(data["amount"]), data.get("description","")))
    conn.commit()
    conn.close()
    return jsonify({"ok": True})
