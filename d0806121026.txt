@app.route("/api/login", methods=["POST"])
def login():
    data = request.json
    conn = get_db()
    key = "admin_pass" if data.get("role") == "admin" else "cashier_pass"
    row = conn.execute("SELECT value FROM settings WHERE key=?", (key,)).fetchone()
    conn.close()
    if row and row["value"] == data.get("password"):
        return jsonify({"ok": True})
    return jsonify({"ok": False}), 401

@app.route("/api/change-password", methods=["POST"])
def change_password():
    data = request.json
    key = "admin_pass" if data.get("role") == "admin" else "cashier_pass"
    conn = get_db()
    conn.execute("UPDATE settings SET value=? WHERE key=?", (data["new_password"], key))
    conn.commit()
    conn.close()
    return jsonify({"ok": True})

@app.route("/api/sales", methods=["POST"])
def add_sale():
    data = request.json
    items = data.get("items", [])
    if not items:
        return jsonify({"ok": False, "error": "No items"}), 400
    conn = get_db()
    total = 0; profit = 0; saved = []
    try:
        for it in items:
            if it["type"] == "lot":
                lot = conn.execute("SELECT * FROM lots WHERE id=?", (it["lid"],)).fetchone()
                if not lot or lot["pieces"] < it["qty"]:
                    return jsonify({"ok": False, "error": it["name"] + " not enough stock!"}), 400
                sub = it["sp"] * it["qty"]
                prf = (it["sp"] - lot["buy"]) * it["qty"]
                total += sub; profit += prf
                conn.execute("UPDATE lots SET pieces=pieces-? WHERE id=?", (it["qty"], it["lid"]))
                saved.append({"type":"lot","ref_id":it["lid"],"name":it["name"],"buy_price":lot["buy"],"sell_price":it["sp"],"qty":it["qty"],"subtotal":sub,"profit":prf})
            else:
                prod = conn.execute("SELECT * FROM products WHERE id=?", (it["pid"],)).fetchone()
                if not prod or prod["stock"] < it["qty"]:
                    return jsonify({"ok": False, "error": it["name"] + " not enough stock!"}), 400
                sub = it["sp"] * it["qty"]
                prf = (it["sp"] - prod["buy_price"]) * it["qty"]
                total += sub; profit += prf
                conn.execute("UPDATE products SET stock=stock-? WHERE id=?", (it["qty"], it["pid"]))
                saved.append({"type":"product","ref_id":it["pid"],"name":it["name"],"buy_price":prod["buy_price"],"sell_price":it["sp"],"qty":it["qty"],"subtotal":sub,"profit":prf})
        now = datetime.now()
        cur = conn.execute("INSERT INTO sales (date,time,total,profit) VALUES (?,?,?,?)", (today(), now.strftime("%I:%M %p"), total, profit))
        sid = cur.lastrowid
        for it in saved:
            conn.execute("INSERT INTO sale_items (sale_id,name,type,ref_id,buy_price,sell_price,qty,subtotal,profit) VALUES (?,?,?,?,?,?,?,?,?)",
                (sid, it["name"], it["type"], it["ref_id"], it["buy_price"], it["sell_price"], it["qty"], it["subtotal"], it["profit"]))
        conn.commit()
        conn.close()
        return jsonify({"ok": True, "total": total, "profit": profit})
    except Exception as e:
        conn.close()
        return jsonify({"ok": False, "error": str(e)}), 500
